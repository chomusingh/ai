from collections import deque

graph = {
    "A": ["B", "C"],
    "B": ["A", "D"],
    "C": ["A", "E"],
    "D": ["B"],
    "E": ["C"]
}

def dfs(node, visited=None):
    if visited is None:          # avoid mutable default argument
        visited = set()
    visited.add(node)
    print(node, end=" ")
    for nei in graph[node]:
        if nei not in visited:
            dfs(nei, visited)

def bfs(start):
    visited, q = set(), deque([start])
    while q:
        node = q.popleft()
        if node not in visited:
            print(node, end=" ")
            visited.add(node)
            q.extend(graph[node])

print("DFS:")
dfs("A")

print("\nBFS:")
bfs("A")
