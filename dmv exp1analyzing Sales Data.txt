import pandas as pd

# Create sample sales data
data = {
    'Product': ['Laptop', 'Mobile', 'Tablet'],
    'Sales': [20000, 15000, 10000]
}

df = pd.DataFrame(data)

# Save data in different formats
df.to_csv('sales.csv', index=False)
df.to_excel('sales.xlsx', index=False)
df.to_json('sales.json', orient='records')

# Read data from different formats
df1 = pd.read_csv('sales.csv')
df2 = pd.read_excel('sales.xlsx')
df3 = pd.read_json('sales.json')

# Combine all data
final = pd.concat([df1, df2, df3], ignore_index=True)

# Display combined data
print("Combined Sales Data:")
print(final)

# Analyze sales
print("\nTotal Sales:", final['Sales'].sum())
print("Average Sales:", final['Sales'].mean())
print("Highest Sales:", final['Sales'].max())

# Product-wise sales
print("\nSales by Product:")
print(final.groupby('Product')['Sales'].sum())
